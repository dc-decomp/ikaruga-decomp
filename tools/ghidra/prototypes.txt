# Prototypes written by hand, applied by ApplyPrototypes.java as user-defined signatures.
#
# The wrappers around the boot ROM's vectors pass their arguments through in r4 and r5 without
# reading them, so the decompiler gives them no parameters, and every caller then drops the
# arguments it passes. These signatures follow the boot ROM's documented calling interface.

int gdGdcGetCmdStat(int id, unsigned int *status);
void gdGdcExecServer(void);
void gdGdcInitSystem(void);
int gdGdcGetDrvStat(unsigned int *status);
int gdGdcG1DmaEnd(void *callback, unsigned int arg);
int gdGdcReqDmaTrans(int id, unsigned int *param);
int gdGdcCheckDmaTrans(int id, unsigned int *size);
int gdGdcReadAbort(int id);
void gdGdcReset(void);
int syBtFlGetInfo(int partition, unsigned int *info);
int syBtFlRead(unsigned int offset, void *buffer, unsigned int size);

int syBtFlDeleteSct(unsigned int offset);
int syBtFntChkSmph(void);
void syBtFntClrSmph(void);

# The C library's single-precision internals. The game is built with SHC's single-precision FPU
# option, so its double is a 4-byte float passed in fr4: as 8-byte doubles every call to them
# decompiled into pairs of halves. The port implements them natively.
float frexp(float value, int *exponent);
float log10(float x);
float _in_atans(float x);
float _in_atan2s(float y, float x);
float _in_fmods(float x, float y);
float _in_ldexps(float x, int exponent);
float _in_log10s(float x);
float _in_logs(float x);

# linkBulletCreate creates an object from four words, two floats and two more words that its callers
# pass on the stack: an angle and a word it keeps. With no parameters there, the C++ read 0 for
# the angle, and stage 3 and 4 objects started at angle 0.
float *linkBulletCreate(unsigned int param_1, unsigned int param_2, unsigned char param_3, float *param_4, float param_5, float param_6, int angle, unsigned int param_8);

# laserLengthToShip turns a line from a player toward a target and returns the line's new length. Its
# caller enemyLaserDraw passes the length in fr4, from the object's word at 0x1c, where the
# decompiler found no parameter, so the C++ started each turn from length 0.
float laserLengthToShip(unsigned int player, int angle, float *target, float length);

# Ninja's linear interpolation of three floats, which the motion state calls through a pointer with
# the frame in fr4. It passes fr4 on to _njGetMotionLinearKeys without reading it, so the
# decompiler gave it no parameter for the frame, and a native version's thunk dropped the frame.
void njMotionIntLinearF3(void *keys, unsigned int count, float *out, float frame);
